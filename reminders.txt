WITH goal AS (
    INSERT INTO savings_goals (user_id, name, target_amount, deadline)
    VALUES (:uid, 'Entrada do apartamento', 30000.00, DATE '2027-12-31')
    RETURNING id
)

INSERT INTO transactions (user_id, category_id, goal_id, amount, transaction_date, description)
SELECT :uid,
       c.id,
       CASE WHEN v.category_name = 'Investimentos' THEN (SELECT id FROM goal) END,
       v.base + ((EXTRACT(MONTH FROM m.month_start)::int * 17 + n * 13) % (v.variation + 1)),
       m.month_start::date + (v.first_day - 1 + (n - 1) * v.interval_days),
       v.description
FROM generate_series(DATE '2025-10-01', DATE '2026-09-01', INTERVAL '1 month') AS m(month_start)
CROSS JOIN (VALUES
    -- categoria,              tipo,      base,   variação, vezes, 1º dia, intervalo, a cada N meses, descrição
    ('Salário',               'INCOME',  5000.00,   0, 1,  5,  0, 1, 'Salário'),
    ('Freelance',             'INCOME',   800.00, 700, 1, 18,  0, 2, 'Projeto freelance'),
    ('Aluguel',               'EXPENSE', 1500.00,   0, 1, 10,  0, 1, 'Aluguel'),
    ('Contas de casa',        'EXPENSE',  220.00, 120, 1, 15,  0, 1, 'Luz, água e internet'),
    ('Mercado',               'EXPENSE',  160.00, 120, 4,  2,  7, 1, 'Mercado'),
    ('Transporte',            'EXPENSE',   35.00,  40, 8,  1,  3, 1, 'Transporte'),
    ('Lazer',                 'EXPENSE',   90.00, 150, 2,  6, 14, 1, 'Lazer'),
    ('Delivery',              'EXPENSE',   45.00,  50, 3,  3,  9, 1, 'Delivery'),
    ('Saúde',                 'EXPENSE',  150.00, 250, 1, 12,  0, 3, 'Consulta ou farmácia'),
    ('Investimentos',         'SAVING',   500.00, 300, 1, 25,  0, 1, 'Aporte: entrada do apartamento'),
    ('Reserva de emergência', 'SAVING',   200.00,   0, 1, 26,  0, 1, 'Aporte na reserva')
) AS v(category_name, category_type, base, variation, times, first_day, interval_days, every_n_months, description)
CROSS JOIN LATERAL generate_series(1, v.times) AS n
JOIN categories c
  ON c.user_id = :uid
 AND c.name = v.category_name
 AND c.type = v.category_type
 AND c.archived = false
WHERE EXTRACT(MONTH FROM m.month_start)::int % v.every_n_months = 0;
